from django.db import models

class Direcciones(models.Model):
    id_direccion = models.AutoField(primary_key=True, db_column='id_direccion')
    calle = models.CharField(max_length=255)
    ciudad = models.CharField(max_length=100)
    estado = models.CharField(max_length=100)
    referencia = models.TextField(blank=True, null=True)

    class Meta:
        managed = False
        db_table = 'direcciones'
        verbose_name = 'Dirección'
        verbose_name_plural = 'Direcciones'

    def __str__(self):
        return f"{self.estado}, {self.ciudad} - {self.calle}"

class ConfiguracionPlantel(models.Model):
    id_config = models.AutoField(primary_key=True, db_column='id_config')
    nombre_plantel = models.CharField(max_length=200)
    codigo_estadistico = models.CharField(max_length=50, blank=True, null=True)
    codigo_plantel = models.CharField(max_length=50, blank=True, null=True)
    id_director_actual = models.IntegerField(blank=True, null=True)
    id_subdirector_actual = models.IntegerField(blank=True, null=True)
    codigo_dea = models.CharField(max_length=50, blank=True, null=True)

    class Meta:
        managed = False
        db_table = 'configuracion_plantel'
        verbose_name = 'Configuración del Plantel'
        verbose_name_plural = 'Configuración del Plantel'

    def __str__(self):
        return self.nombre_plantel

       writer.writerow([

           f"{relacion.representante.primer_nombre} {relacion.representante.primer_apellido}",

           relacion.parentesco or 'No especificado',
           'Sí' if relacion.es_principal else 'No'
       ])

    return response

@api_view(['GET'])

def exportar_alumnos_representantes_excel(request):
    wb = openpyxl.Workbook()
    ws = wb.active
    ws.title = "Alumnos y Representantes"

    fuente_titulo = Font(name="Arial", size=11, bold=True, color="FFFFFF")
    relleno_azul = PatternFill(start_color="1F4E78", end_color="1F4E78", fill_type="solid")

    encabezados = ['Alumno', 'Representante', 'Teléfono', 'Parentesco', 'Es Principal']
    ws.append(encabezados)

    for col_num in range(1, 6):
        celda = ws.cell(row=1, column=col_num)
        celda.font = fuente_titulo
        celda.fill = relleno_azul

    relaciones = AlumnoRepresentante.objects.select_related('alumno', 'representante')

    for relacion in relaciones:
        ws.append([
            f"{relacion.alumno.primer_nombre} {relacion.alumno.primer_apellido}",

            relacion.parentesco or 'No especificado',
            "Sí" if relacion.es_principal else "No"
        ])

    response = HttpResponse(
        content_type='application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'
    )
    response['Content-Disposition'] = 'attachment; filename="reporte_institucional.xlsx"'

    wb.save(response)
    return response

class EsProfesorOAdmin(permissions.BasePermission):
    """
    Permiso personalizado que solo permite acceso a usuarios con rol 'profesor' o 'admin'.
    """

    def has_permission(self, request, view):
        if request.user and hasattr(request.user, 'rol'):
            return request.user.rol in ['profesor', 'admin']
        return False

    def has_object_permission(self, request, view, obj):
        if request.user and hasattr(request.user, 'rol'):
            return request.user.rol in ['profesor', 'admin']
        return False

8. CARACTERÍSTICAS ESPECIALES
================================================================================

SISTEMA DE ASISTENCIAS:
- Estados: P (Presente), A (Ausente), T (Tarde), J (Justificado)
- Restricción única: Un alumno solo puede tener un registro por fecha y grupo
- Profesores pueden registrar y actualizar asistencias
- Estadísticas por grupo y por alumno
- Porcentaje de asistencia calculado automáticamente
- Exportación en CSV, Excel y PDF con filtros por grupo y rango de fechas

EXPORTACIONES:
- Personal: Exporta alumnos y representantes con nombres, teléfonos, parentesco
- Asistencias: Exporta registros con nombre alumno, cédula escolar, grado, sección, estado, observaciones
- Formatos: CSV, Excel (con estilos), PDF (con diseño institucional)
- Filtros: Por grupo, rango de fechas

PERMISOS:
- EsProfesorOAdmin: Solo permite acceso a usuarios con rol 'profesor' o 'admin'
- Exportaciones: Permisos AllowAny (acceso público)

VALIDACIONES:
- Calificaciones: Solo acepta 'A', 'B', 'C', 'D'
- Asistencias: Solo acepta 'P', 'A', 'T', 'J'

BASE DE DATOS:
- managed = False (Django no modifica la estructura de la base de datos)
- Tablas existentes creadas con script SQL externo
- Mapeo explícito de columnas con db_column

9. INSTRUCCIONES DE INSTALACIÓN Y EJECUCIÓN
================================================================================

1. Crear entorno virtual:
   python -m venv venv
   source venv/bin/activate  # Linux/Mac
   venv\Scripts\activate  # Windows

2. Instalar dependencias:
   pip install -r requirements.txt

3. Configurar archivo .env con credenciales de base de datos

4. Crear base de datos y usuario en MariaDB:
   CREATE DATABASE plataforma_escolar CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
   CREATE USER 'admin_plataforma'@'localhost' IDENTIFIED BY 'tu_contraseña';
   GRANT ALL PRIVILEGES ON plataforma_escolar.* TO 'admin_plataforma'@'localhost';
   FLUSH PRIVILEGES;

5. Ejecutar script SQL de tablas:
   mariadb -u admin_plataforma -p plataforma_escolar < planter_educativo_v12.sql

6. Crear tabla de asistencias:
   mariadb -u admin_plataforma -p plataforma_escolar < crear_tabla_asistencias.sql

7. Ejecutar migraciones:
   python manage.py makemigrations
   python manage.py migrate

8. Poblar datos de prueba:
   python poblar_datos_prueba.py

9. Ejecutar servidor:
   python manage.py runserver

10. Acceder a Swagger UI:
    http://localhost:8000/swagger/

10. ESTRUCTURA DE LA BASE DE DATOS
================================================================================

TABLAS PRINCIPALES:
- configuracion_plantel: Datos generales del plantel
- periodos_academicos: Catálogo de períodos escolares
- niveles_educativos: Catálogo de niveles (Primaria, Secundaria, etc.)
- turnos: Catálogo de turnos (Mañana, Tarde)
- grados_academicos: Grados por nivel
- secciones: Secciones (A, B, C, etc.)
- direcciones: Direcciones físicas
- usuarios: Usuarios del sistema con roles
- profesores: Información docente
- alumnos: Datos detallados de estudiantes
- representantes_legales: Información de representantes
- alumno_representante: Relación alumno-representante
- grupos_academicos: Combinación grado/sección/turno/periodo
- inscripciones: Inscripción de alumnos en grupos
- calificaciones: Calificaciones (A, B, C, D)
- asistencias: Registro diario de asistencia (P, A, T, J)
