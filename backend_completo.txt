from rest_framework import permissions

class EsAdministrador(permissions.BasePermission):
    """
    Permite acceso solo a usuarios con rol 'admin' o superusuarios.
    """
    def has_permission(self, request, view):
        if not request.user or not request.user.is_authenticated:
            return False
        return getattr(request.user, 'is_superuser', False) or getattr(request.user, 'rol', None) == 'admin'


class EsProfesor(permissions.BasePermission):
    """
    Permite acceso a usuarios con rol 'profesor'.
    """
    def has_permission(self, request, view):
        if not request.user or not request.user.is_authenticated:
            return False
        return getattr(request.user, 'rol', None) == 'profesor'


class EsProfesorOAdmin(permissions.BasePermission):
    """
    Permite acceso de lectura/escritura a Administradores y Profesores.
    """
    def has_permission(self, request, view):
        if not request.user or not request.user.is_authenticated:
            return False
        rol = getattr(request.user, 'rol', None)
        return getattr(request.user, 'is_superuser', False) or rol in ['admin', 'profesor', 'directivo']

class AlumnosViewSet(viewsets.ModelViewSet):
    queryset = Alumnos.objects.select_related('direccion').all()
    serializer_class = AlumnosSerializer
    permission_classes = [EsProfesorOAdmin]
    filter_backends = [filters.SearchFilter, filters.OrderingFilter]
    search_fields = ['cedula_escolar', 'cedula_identidad', 'nombres', 'apellidos']
    ordering_fields = ['apellidos', 'nombres', 'created_at']

    @action(detail=True, methods=['get'])
    def historial_inscripciones(self, request, pk=None):
        alumno = self.get_object()
        inscripciones = Inscripciones.objects.filter(alumno=alumno).select_related('periodo', 'grupo_academico')
        serializer = InscripcionesSerializer(inscripciones, many=True)
        return Response(serializer.data)


class GruposAcademicosViewSet(viewsets.ModelViewSet):
    queryset = GruposAcademicos.objects.select_related('periodo', 'grado', 'seccion', 'turno', 'profesor_guia').all()
    serializer_class = GruposAcademicosSerializer
    permission_classes = [EsProfesorOAdmin]
    filterset_fields = ['periodo', 'grado', 'seccion', 'turno', 'profesor_guia']


class InscripcionesViewSet(viewsets.ModelViewSet):
    queryset = Inscripciones.objects.select_related('alumno', 'grupo_academico', 'periodo').all()
    serializer_class = InscripcionesSerializer
    permission_classes = [EsProfesorOAdmin]
    filterset_fields = ['periodo', 'estado', 'alumno', 'grupo_academico']
    search_fields = ['alumno__nombres', 'alumno__apellidos', 'alumno__cedula_escolar']


class CalificacionesViewSet(viewsets.ModelViewSet):
    queryset = Calificaciones.objects.select_related('inscripcion__alumno', 'profesor').all()
    serializer_class = CalificacionesSerializer
    permission_classes = [EsProfesorOAdmin]
    filterset_fields = ['inscripcion', 'materia_asignatura', 'lapso_periodo', 'profesor']
    search_fields = ['materia_asignatura', 'inscripcion__alumno__nombres', 'inscripcion__alumno__apellidos']

    @action(detail=False, methods=['get'])
    def por_alumno(self, request):
        alumno_id = request.query_params.get('alumno_id')
        if not alumno_id:
            return Response({'error': 'Parámetro alumno_id es requerido'}, status=status.HTTP_400_BAD_REQUEST)
        
        calificaciones = Calificaciones.objects.filter(inscripcion__alumno_id=alumno_id)
        serializer = self.get_serializer(calificaciones, many=True)
        return Response(serializer.data)

from rest_framework.permissions import BasePermission, SAFE_METHODS


class IsAdministrador(BasePermission):
    """
    Permiso exclusivo para usuarios con rol de Administrador.
    """
    def has_permission(self, request, view):
        if not (request.user and request.user.is_authenticated):
            return False
        return hasattr(request.user, 'rol') and request.user.rol == 'ADMINISTRADOR'


class IsDocenteOrAdmin(BasePermission):
    """
    Permite acceso total a Administradores y lectura/escritura de expedientes a Profesores/Docentes.
    """
    def has_permission(self, request, view):
        if not (request.user and request.user.is_authenticated):
            return False
        
        # Permitir métodos de lectura a cualquier usuario autenticado
        if request.method in SAFE_METHODS:
            return True
        
        # Permitir escritura si es Docente o Administrador
        user_rol = getattr(request.user, 'rol', None)
        return user_rol in ['PROFESOR', 'ADMINISTRADOR', 'DIRECTIVO']

class AlumnoRegistroCompletoSerializer(serializers.ModelSerializer):
    direccion = DireccionSerializer(required=False, write_only=True)
    representante = RepresentanteLegalSerializer(required=False, write_only=True)
    parentesco = serializers.CharField(max_length=50, required=False, write_only=True, default='Padre/Madre')
    es_principal = serializers.BooleanField(required=False, write_only=True, default=True)

    class Meta:
        model = Alumno
        fields = [
            'id_alumno', 'primer_nombre', 'segundo_nombre', 'primer_apellido', 'segundo_apellido',
            'tipo_documento_identidad', 'numero_documento_identidad', 'cedula_escolar',
            'fecha_nacimiento', 'genero', 'lugar_nacimiento', 'nacionalidad',
            'entidad_federal_nacimiento', 'municipio_nacimiento', 'parroquia_nacimiento',
            'condicion_fisica', 'grupo_sanguineo', 'apoyo_escolar', 'vacunas_completas',
            'alergico_a', 'tratamiento_medico', 'peso_kg', 'talla_cm', 'talla_camisa',
            'talla_pantalon', 'talla_zapato', 'estado_civil_padres', 'ingreso_familiar',
            'profesion_madre', 'profesion_padre', 'tipo_vivienda', 'numero_hermanos',
            'numero_en_el_planter', 'discapacidad', 'tipo_discapacidad', 'observaciones',
            'fecha_inscripcion_inicial', 'estatus_general', 'direccion', 'representante',
            'parentesco', 'es_principal'
        ]

    def validate_numero_documento_identidad(self, value):
        if Alumno.objects.filter(numero_documento_identidad=value).exists():
            raise serializers.ValidationError("Un estudiante con este número de documento ya está registrado.")
        return value

    @transaction.atomic
    def create(self, validated_data):
        direccion_data = validated_data.pop('direccion', None)
        representante_data = validated_data.pop('representante', None)
        parentesco = validated_data.pop('parentesco', 'Padre/Madre')
        es_principal = validated_data.pop('es_principal', True)

        # 1. Crear Dirección de Residencia si aplica
        direccion_obj = None
        if direccion_data:
            direccion_obj = Direccion.objects.create(**direccion_data)
            validated_data['id_direccion_residencia'] = direccion_obj

        # 2. Crear Alumno
        alumno = Alumno.objects.create(**validated_data)

        # 3. Crear o Vincular Representante Legal
        if representante_data:
            cedula_rep = representante_data.get('cedula')
            rep_obj = RepresentanteLegal.objects.filter(cedula=cedula_rep).first()
            if not rep_obj:
                rep_obj = RepresentanteLegal.objects.create(**representante_data)

            # 4. Crear relación Alumno - Representante
            AlumnoRepresentante.objects.create(
                id_alumno=alumno,
                id_representante=rep_obj,
                parentesco=parentesco,
                es_principal=es_principal
            )

        return alumno

class AlumnoViewSet(viewsets.ModelViewSet):
    """
    ViewSet para la gestión de alumnos y sus expedientes completos.
    Soporta filtros por género, estatus, cédula, búsqueda por nombre y ordenamiento.
    """
    queryset = Alumno.objects.all().select_related('id_direccion_residencia').order_by('primer_apellido', 'primer_nombre')
    permission_classes = [permissions.IsAuthenticatedOrReadOnly]
    filter_backends = [DjangoFilterBackend, filters.SearchFilter, filters.OrderingFilter]
    filterset_fields = [
        'genero', 'estatus_general', 'tipo_documento_identidad',
        'apoyo_escolar', 'vacunas_completas', 'discapacidad'
    ]
    search_fields = [
        'numero_documento_identidad', 'cedula_escolar',
        'primer_nombre', 'segundo_nombre', 'primer_apellido', 'segundo_apellido'
    ]
    ordering_fields = ['primer_apellido', 'fecha_nacimiento', 'fecha_inscripcion_inicial']

    def get_serializer_class(self):
        if self.action in ['create', 'update', 'partial_update']:
            return AlumnoRegistroCompletoSerializer
        return AlumnoDetalleSerializer

    @action(detail=True, methods=['get'])
    def historial_inscripciones(self, request, pk=None):
        alumno = self.get_object()
        inscripciones = Inscripcion.objects.filter(id_alumno=alumno).select_related('id_grupo')
        serializer = InscripcionSerializer(inscripciones, many=True)
        return Response(serializer.data)

    @action(detail=False, methods=['get'])
    def resumen_antropometrico(self, request):
        alumnos = self.get_queryset()
        total = alumnos.count()
        sin_vacunas = alumnos.filter(vacunas_completas=False).count()
        con_discapacidad = alumnos.filter(discapacidad=True).count()
        con_apoyo = alumnos.filter(apoyo_escolar=True).count()

        return Response({
            'total_estudiantes': total,
            'sin_vacunas_completas': sin_vacunas,
            'con_discapacidad': con_discapacidad,
            'con_apoyo_escolar': con_apoyo
        })

    def __str__(self):
        return f"{self.primer_nombre} {self.primer_apellido} - C.I. {self.cedula}"


class AlumnoRepresentante(models.Model):
    """
    Tabla relacional pivote entre Alumno y Representante Legal.
    Refleja clave primaria compuesta (id_alumno, id_representante).
    En Django con managed = False, se establece primary_key=True en la relación
    con 'alumno' y unique_together en la Meta para soportar la integridad compuesta.
    """
    alumno = models.ForeignKey(
        'academico.Alumnos',
        on_delete=models.CASCADE,
        db_column='id_alumno',
        primary_key=True
    )
    representante = models.ForeignKey(
        RepresentantesLegales,
        on_delete=models.CASCADE,
        db_column='id_representante'
    )
    parentesco = models.CharField(
        max_length=50,
        blank=True,
        null=True,
        db_column='parentesco'
    )
    es_principal = models.BooleanField(
        default=True,
        db_column='es_principal'
    )

    class Meta:
        managed = False
        db_table = 'alumno_representante'
        unique_together = (('alumno', 'representante'),)
        verbose_name = 'Asociación Alumno - Representante'
        verbose_name_plural = 'Asociaciones Alumnos - Representantes'

    def __str__(self):
        return f"Alumno ID {self.alumno_id} <-> Representante ID {self.representante_id}"

# Plataforma Escolar Backend (Django + MariaDB)

Este proyecto proporciona la estructura inicial Backend en Django para la gestión escolar, conectada a una base de datos MariaDB existente (`plataforma_escolar`).

## Estructura de Aplicaciones

- `apps.core`: Configuración del plantel, períodos académicos, niveles, turnos, grados, secciones, direcciones y usuarios.
- `apps.personal`: Profesores, representantes legales y la relación pivot `alumno_representante`.
- `apps.academico`: Alumnos, grupos académicos, inscripciones y calificaciones.

## Características Clave

1. Modelos con `managed = False` mapeados directamente a la base de datos MariaDB `plataforma_escolar`.
2. Conexión de motor `django.db.backends.mysql`.
3. Vistas y Serializers DRF para Alumnos, Inscripciones, Calificaciones, Profesores y más.
4. Permisos personalizados por rol (`EsAdministrador`, `EsProfesor`, `EsProfesorOAdmin`).
